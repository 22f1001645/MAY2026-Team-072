# 03-01: Tracer — Submission Guards, Budget Cap & Team Roster Validation

**Phase**: 3 — Proposal Collaboration, Team Roster & Budget Guardrails
**Requirements**: PROPOSAL-00, PROPOSAL-02, BUDGET-01
**Depends on**: Phase 2 complete

## Goal

Prove that `POST /api/v1/activities` now enforces (1) submitter holds an active organiser POR **in the specific primary org** they propose for, (2) budget total is hard-capped at `min(capacity × ₹200, ₹20,000)` for meetup scales, and (3) a team roster with exactly one Meetup Head is required — with all three verified end-to-end by pytest before any other Phase 3 work touches the submission path.

## Tracer Scope

The tracer is the enriched `submit_proposal` service function plus the three validation layers. It does **not** include collaborator rows (Plan 03-02), response endpoints (03-02), admin queue enrichment (03-03), or any frontend changes (03-04).

```python
def has_organiser_por_in_org(user: User, org) -> bool:
    """PROPOSAL-00: does user hold an active ORGANISER_PORS POR in this specific org?

    Checks ORGANISER_PORS (not ORG_MANAGER_PORS) because PROPOSAL-00 explicitly
    includes Event Lead — an Event Lead is in ORGANISER_PORS but not ORG_MANAGER_PORS.
    The boundary is org ID identity — two organisations that share a name or city
    are still two organisations."""
    if not user or not org:
        return False
    for por in _active_pors(user):
        if por.por_type in ORGANISER_PORS and org.id in _por_org_ids(por):
            return True
    return False
```

```python
def submit_proposal(data: dict, organiser) -> Activity:
    org = OrgRepository.by_id(data["org_id"])
    if not org:
        raise ApiError("Organisation not found", "org_not_found", 404)

    # PROPOSAL-00: submitter must hold an active organiser POR in the primary org
    if not has_organiser_por_in_org(organiser, org):
        raise ApiError(
            "You must hold an active organiser POR in this organisation to propose on its behalf",
            "forbidden",
            403,
        )

    # BUDGET-01: hard cap for meetup scales (regional_fest is sponsor-funded, exempt)
    if data["scale"] in ("local_meetup", "regional_meetup"):
        requested = sum(line["planned"] for line in data.get("budget", []))
        allowed = min(data["capacity"] * BUDGET_CAPACITY_RATE, BUDGET_ABSOLUTE_CEILING)
        if requested > allowed:
            raise ApiError(
                f"Budget ₹{requested:,} exceeds the cap of ₹{allowed:,} "
                f"for a {data['capacity']}-person {data['scale'].replace('_', ' ')} "
                f"(₹{BUDGET_CAPACITY_RATE}/person, max ₹{BUDGET_ABSOLUTE_CEILING:,})",
                "budget_cap_exceeded",
                400,
            )

    # PROPOSAL-02: validate team roster — exactly one Meetup Head required
    team = data.get("team", [])
    meetup_heads = [m for m in team if m["role"] == TEAM_ROLE_MEETUP_HEAD]
    if len(meetup_heads) == 0:
        raise ApiError(
            "A proposal must name exactly one Meetup Head",
            "invalid_team",
            400,
        )
    if len(meetup_heads) > 1:
        raise ApiError(
            "A proposal must name exactly one Meetup Head",
            "invalid_team",
            400,
        )
    for member in team:
        if member["role"] not in TEAM_ROLES:
            raise ApiError(f"Invalid team role: {member['role']!r}", "invalid_role", 400)
        if not User.query.get(member["user_id"]):
            raise ApiError(f"Team member not found: {member['user_id']!r}", "user_not_found", 404)

    # Quota guardrail (FR-B2): one meetup per region per month
    if data["scale"] in ("local_meetup", "regional_meetup"):
        count = ActivityRepository.meetups_in_region_month(data["org_id"], data["starts_at"])
        if count >= MEETUPS_PER_CITY_PER_MONTH:
            raise ApiError(
                "Quota guardrail: this region already has a meetup this month",
                "quota_exceeded",
                409,
            )

    activity = Activity(
        org_id=org.id,
        title=data["title"],
        description=data.get("description", ""),
        city=data["city"],
        scale=data["scale"],
        venue=data["venue"],
        starts_at=data["starts_at"],
        capacity=data["capacity"],
        status=EVENT_UPCOMING,
        organiser_id=organiser.id,
    )
    db.session.add(activity)
    db.session.flush()

    for line in data.get("budget", []):
        db.session.add(
            BudgetLine(activity_id=activity.id, label=line["label"], planned=line["planned"])
        )

    # PROPOSAL-02: write team roster
    for member in team:
        db.session.add(ActivityTeamMember(
            activity_id=activity.id,
            user_id=member["user_id"],
            role=member["role"],
            added_by=organiser.id,
        ))

    proposal = Proposal(activity_id=activity.id, submitted_by=organiser.id, status=PROPOSAL_SUBMITTED)
    db.session.add(proposal)

    AuditRepository.record("proposal", activity.id, "Submitted proposal", organiser)
    db.session.commit()
    return activity
```

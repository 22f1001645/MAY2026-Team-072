```tsx
function QRTicket({ registration }: { registration: any }) {
  const a = registration.activity
  if (!a || registration.status !== 'confirmed') return null
  const isPast = new Date(a.starts_at) < new Date()

  return (
    <div className="flex flex-col items-center gap-2 py-4 border-b border-divider/50">
      <div className="text-caption uppercase tracking-wider text-graytext-light">
        Your Check-In Code
      </div>
      {registration.attended ? (
        <Badge tone="success">Attended ✓</Badge>
      ) : isPast ? (
        <Badge tone="neutral">Event Ended</Badge>
      ) : (
        <>
          <QRCodeCanvas value={registration.id} size={180}
            bgColor="transparent" fgColor="#5CE6C9" level="M" />
          <div className="text-caption text-graytext-mid">
            Show this to the organiser at the door
          </div>
        </>
      )}
    </div>
  )
}
```

Insert `<QRTicket registration={r} />` above each registration card in the filtered list.

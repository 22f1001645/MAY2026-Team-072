```ts
// Phase 6 hooks
export const useTimeline = (id?: string) =>
  useQuery({ queryKey: ['timeline', id], queryFn: () => api.timeline(id!), enabled: !!id })

export const useScanQrMutation = () =>
  useMutation({
    mutationFn: ({ activityId, registrationId }: { activityId: string; registrationId: string }) =>
      api.scanQr(activityId, registrationId),
  })

export const useDraftExpensesMutation = () => {
  const qc = useQueryClient()
  return useMutation({
    mutationFn: ({ activityId, body }: { activityId: string; body: object }) =>
      api.draftExpenses(activityId, body as any),
    onSuccess: (_, { activityId }) => qc.invalidateQueries({ queryKey: ['expenses', activityId] }),
  })
}

export const usePatchReportMutation = () => {
  const qc = useQueryClient()
  return useMutation({
    mutationFn: ({ activityId, body }: { activityId: string; body: object }) =>
      api.patchReport(activityId, body as any),
    onSuccess: (_, { activityId }) => qc.invalidateQueries({ queryKey: ['report', activityId] }),
  })
}

export const useSubmitFeedbackMutation = () => {
  const qc = useQueryClient()
  return useMutation({
    mutationFn: ({ activityId, body }: { activityId: string; body: { rating: number; comment?: string } }) =>
      api.submitFeedback(activityId, body),
    onSuccess: (_, { activityId }) => qc.invalidateQueries({ queryKey: ['my-feedback', activityId] }),
  })
}

export const useMyFeedback = (id?: string) =>
  useQuery({ queryKey: ['my-feedback', id], queryFn: () => api.myFeedback(id!), enabled: !!id })
```

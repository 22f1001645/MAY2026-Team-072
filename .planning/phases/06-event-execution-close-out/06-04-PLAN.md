from sqlalchemy.exc import IntegrityError
from ..errors import ApiError
from ..extensions import db
from ..models import EventFeedback, Registration

def submit(activity_id, data, user):
    """FEEDBACK-02: Attendee rates an event. Gated on Attendance record."""
    reg = Registration.query.filter_by(
        activity_id=activity_id, student_id=user.id
    ).first()
    if not reg or reg.attendance is None:
        raise ApiError("Only attendees can submit feedback", "not_attendee", 403)

    feedback = EventFeedback(
        activity_id=activity_id, user_id=user.id,
        rating=data["rating"], comment=data.get("comment")
    )
    db.session.add(feedback)
    try:
        db.session.commit()
    except IntegrityError:
        db.session.rollback()
        raise ApiError("Feedback already submitted", "duplicate_feedback", 409)

    return feedback.to_dict()

def get_mine(activity_id, user):
    fb = EventFeedback.query.filter_by(
        activity_id=activity_id, user_id=user.id
    ).first()
    if fb:
        return {**fb.to_dict(), "submitted": True}
    return {"submitted": False}
```

from flask import Blueprint, g, jsonify, request
from ..rbac import auth_required
from ..schemas import FeedbackSchema
from ..services import feedback_service

feedback_bp = Blueprint("feedback", __name__, url_prefix="/api/v1")

@feedback_bp.post("/activities/<activity_id>/feedback")
@auth_required
def submit_feedback(activity_id):
    data = FeedbackSchema().load(request.get_json() or {})
    result = feedback_service.submit(activity_id, data, g.current_user)
    return jsonify(result), 201

@feedback_bp.get("/activities/<activity_id>/feedback/me")
@auth_required
def my_feedback(activity_id):
    result = feedback_service.get_mine(activity_id, g.current_user)
    return jsonify(result), 200
```

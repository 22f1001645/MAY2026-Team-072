```tsx
function FeedbackForm({ activityId, registration }: { activityId: string; registration: any }) {
  const { data: existing } = useMyFeedback(activityId)
  const submitMut = useSubmitFeedbackMutation()
  const { addToast } = useToast()
  const [rating, setRating] = useState(0)
  const [comment, setComment] = useState('')

  // Gate: must have attended + event must be past
  if (!registration?.attended) return null
  const eventPast = new Date(registration.activity?.starts_at) < new Date()
  if (!eventPast) return null

  // Already submitted state
  if (existing?.submitted || submitMut.isSuccess) {
    return (
      <Card className="p-6 mb-8">
        <Badge tone="success">Feedback Submitted</Badge>
        <p className="mt-2 text-body text-graytext-light">
          Thank you! Your rating: {'★'.repeat(existing?.rating ?? rating)}
        </p>
      </Card>
    )
  }

  const handleSubmit = () => {
    submitMut.mutate(
      { activityId, body: { rating, comment: comment || undefined } },
      { onSuccess: () => addToast('Feedback submitted — thank you!') }
    )
  }

  return (
    <Card className="p-6 mb-8">
      <h3 className="text-h3 text-offwhite mb-4">Rate This Event</h3>
      <div role="radiogroup" aria-label="Event rating" className="flex gap-2 mb-4">
        {[1,2,3,4,5].map(n => (
          <button key={n} role="radio" aria-label={`${n} star${n>1?'s':''}`}
            aria-checked={rating === n}
            onClick={() => setRating(r => r === n ? 0 : n)}
            className={`text-2xl cursor-pointer transition-colors ${
              n <= rating ? 'text-[#FFD700]' : 'text-divider hover:text-graytext-mid'
            }`}>
            ★
          </button>
        ))}
      </div>
      <textarea className="w-full rounded-lg bg-surface-2 border border-divider p-3 text-body text-offwhite placeholder:text-graytext-mid focus:border-teal focus:outline-none resize-none"
        placeholder="Share your thoughts (optional)" rows={3}
        value={comment} onChange={e => setComment(e.target.value)} />
      <Button className="mt-4" disabled={rating === 0 || submitMut.isPending}
        onClick={handleSubmit}>
        {submitMut.isPending ? 'Submitting…' : 'Submit Feedback'}
      </Button>
    </Card>
  )
}
```

# 03-04: Frontend — Proposal Form, Approval Queue & Collaborator Respond UI

**Phase**: 3 — Proposal Collaboration, Team Roster & Budget Guardrails
**Requirements**: PROPOSAL-01, PROPOSAL-02, PROPOSAL-03, PROPOSAL-04, BUDGET-01 (UI surfaces)
**Depends on**: 03-03-PLAN.md
**UI hint**: yes

## Goal

Prove that `ProposalNew.tsx` lets an organiser attach collaborating orgs and a team roster with live budget cap feedback, that `ApprovalQueue.tsx` shows collaborators and team members alongside the existing proposal fields, and that the collaborator respond UI lets an authority holder accept or decline an invite from the event detail page.

## Tracer Scope

**In scope**: Proposal form extension (collaborator picker + team roster builder + budget cap indicator), approval queue enriched columns, collaborator respond buttons on EventDetail, new TypeScript types, and vitest tests for the client-side logic.  
**Not in scope**: Campaign push-live (Phase 5), org profile page (Phase 4), QR check-in (Phase 6).

```typescript
export async function respondCollaboratorInvite(
  activityId: string,
  collabId: string,
  status: 'accepted' | 'declined',
): Promise<Collaborator> {
  return apiFetch(`/activities/${activityId}/collaborators/${collabId}/respond`, {
    method: 'POST',
    body: JSON.stringify({ status }),
  })
}

export async function reassignMeetupHead(
  activityId: string,
  newMeetupHeadUserId: string,
): Promise<{ activity_id: string; new_meetup_head_user_id: string }> {
  return apiFetch(`/activities/${activityId}/reassign-meetup-head`, {
    method: 'POST',
    body: JSON.stringify({ new_meetup_head_user_id: newMeetupHeadUserId }),
  })
}
```

In `queries.ts`, add:

```typescript
export function useRespondCollabInvite() {
  const qc = useQueryClient()
  return useMutation({
    mutationFn: ({
      activityId,
      collabId,
      status,
    }: {
      activityId: string
      collabId: string
      status: 'accepted' | 'declined'
    }) => respondCollaboratorInvite(activityId, collabId, status),
    onSuccess: (_, { activityId }) => {
      qc.invalidateQueries({ queryKey: ['activity', activityId] })
      qc.invalidateQueries({ queryKey: ['proposals'] })
    },
  })
}
```

```typescript
import { describe, it, expect } from 'vitest'

function computeBudgetCap(scale: string, capacity: number): number | null {
  if (scale === 'local_meetup' || scale === 'regional_meetup') {
    return Math.min(capacity * 200, 20_000)
  }
  return null // exempt
}

describe('Budget cap computation (BUDGET-01)', () => {
  it('local_meetup capacity=50 → cap=10000', () => {
    expect(computeBudgetCap('local_meetup', 50)).toBe(10_000)
  })

  it('local_meetup capacity=200 → cap=20000 (absolute ceiling)', () => {
    expect(computeBudgetCap('local_meetup', 200)).toBe(20_000)
  })

  it('regional_meetup capacity=100 → cap=20000', () => {
    expect(computeBudgetCap('regional_meetup', 100)).toBe(20_000)
  })

  it('regional_fest is exempt → null', () => {
    expect(computeBudgetCap('regional_fest', 500)).toBeNull()
  })

  it('budget at cap is valid', () => {
    const cap = computeBudgetCap('local_meetup', 50)!
    expect(10_000 <= cap).toBe(true)
  })

  it('budget over cap is invalid', () => {
    const cap = computeBudgetCap('local_meetup', 50)!
    expect(10_001 > cap).toBe(true)
  })
})
```

```typescript
import { describe, it, expect } from 'vitest'

// Pure logic test — is_mega_meetup derivation on the client matches the backend rule
function isMegaMeetup(collaborators: Array<{ status: string }>): boolean {
  return collaborators.filter(c => c.status === 'accepted').length >= 3
}

describe('Mega Meetup label (PROPOSAL-03)', () => {
  it('False with 2 accepted collaborators', () => {
    const collabs = [
      { status: 'accepted' },
      { status: 'accepted' },
      { status: 'pending' },
    ]
    expect(isMegaMeetup(collabs)).toBe(false)
  })

  it('True with 3 accepted collaborators', () => {
    const collabs = [
      { status: 'accepted' },
      { status: 'accepted' },
      { status: 'accepted' },
    ]
    expect(isMegaMeetup(collabs)).toBe(true)
  })

  it('Declined collaborators do not count', () => {
    const collabs = [
      { status: 'accepted' },
      { status: 'accepted' },
      { status: 'declined' },
    ]
    expect(isMegaMeetup(collabs)).toBe(false)
  })
})
```

2. Add query management functions to `backend/app/services/channel_service.py`:

- `submit_query(activity_id, body, current_user)`:
  - Verify activity exists and call `_assert_channel_open(activity)`.
  - Verify caller is registered attendee (`RegistrationRepository.find(activity_id, current_user.id)` is not None) or organiser.
  - Create `EventQuery(activity_id=activity_id, user_id=current_user.id, body=body, status=QUERY_OPEN)`.
  - Commit and return query.

- `list_queries(activity_id)`:
  - Query `EventQuery.query.filter_by(activity_id=activity_id).order_by(EventQuery.created_at.desc()).all()`.
  - Return list of queries.

- `my_queries(activity_id, current_user)`:
  - Query `EventQuery.query.filter_by(activity_id=activity_id, user_id=current_user.id).order_by(EventQuery.created_at.desc()).all()`.
  - Return list of queries.

- `respond_query(activity_id, query_id, response_body, current_user)`:
  - Fetch query by `id=query_id` and `activity_id=activity_id`. If not found, raise 404.
  - If `query.status != QUERY_OPEN`: raise `ApiError("Query is already answered", "already_answered", 409)`.
  - Set `query.response_body = response_body`, `query.responded_by = current_user.id`, `query.responded_at = datetime.now(timezone.utc)`, `query.status = QUERY_ANSWERED`.
  - Commit and return query.

from ..schemas import PollCreateSchema, PollVoteSchema, QuerySubmitSchema, QueryRespondSchema

@channel_bp.post("/events/<activity_id>/queries")
@auth_required
def submit_query(activity_id):
    data = QuerySubmitSchema().load(request.get_json() or {})
    query = channel_service.submit_query(activity_id, data["body"], g.current_user)
    return jsonify(query.to_dict()), 201

@channel_bp.get("/events/<activity_id>/queries")
@can_manage_activity_required
def list_queries(activity_id):
    queries = channel_service.list_queries(activity_id)
    return jsonify([q.to_dict() for q in queries]), 200

@channel_bp.get("/events/<activity_id>/queries/mine")
@auth_required
def my_queries(activity_id):
    queries = channel_service.my_queries(activity_id, g.current_user)
    return jsonify([q.to_dict() for q in queries]), 200

@channel_bp.post("/events/<activity_id>/queries/<query_id>/respond")
@can_manage_activity_required
def respond_query(activity_id, query_id):
    data = QueryRespondSchema().load(request.get_json() or {})
    query = channel_service.respond_query(activity_id, query_id, data["response_body"], g.current_user)
    return jsonify(query.to_dict()), 200

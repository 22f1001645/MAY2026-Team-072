export default function CheckIn() {

        right={<Badge tone="teal">{attended} / {roster.length} in</Badge>} />
      {/* Tab switcher */}
      <div role="tablist" className="flex gap-2 mb-6">
        <Button role="tab" aria-selected={tab==='qr'} ... >QR Scanner</Button>
        <Button role="tab" aria-selected={tab==='roster'} ... >Roster</Button>
      </div>
      {tab === 'qr' ? <QRScannerTab /> : <RosterTab />}
    </div>
  )
}

```python
def draft_and_submit(activity_id, rows, submit, social_post_url, user):
    """REIMB-01/02: Bulk upsert expense drafts; Meetup Head submit with social URL."""
    activity = ActivityRepository.by_id(activity_id)
    if not activity:
        raise ApiError("Event not found", "not_found", 404)

    # Upsert draft rows
    result_expenses = []
    for row in rows:
        if row.get("id"):
            expense = Expense.query.get(row["id"])
            if expense and expense.activity_id == activity_id:
                expense.label = row.get("label", expense.label)
                expense.amount = row.get("amount", expense.amount)
                expense.bill_name = row.get("bill_name", expense.bill_name)
        else:
            expense = Expense(activity_id=activity_id, label=row["label"],
                              amount=row.get("amount", 0), bill_name=row.get("bill_name"),
                              status=EXPENSE_DRAFT, submitted_by=user.id)
            db.session.add(expense)
        result_expenses.append(expense)

    if submit:
        # Meetup Head gate
        is_head = ActivityTeamMember.query.filter_by(
            activity_id=activity_id, user_id=user.id, role=TEAM_ROLE_MEETUP_HEAD
        ).first()
        if not is_head and user.role != ROLE_ADMIN:
            raise ApiError("Only the Meetup Head can submit", "forbidden", 403)
        if not social_post_url:
            raise ApiError("Social post URL is required", "missing_social_url", 422)
        # Validate URL safety
        # ... url_safety.check(social_post_url)
        # Transition all drafts to submitted
        drafts = Expense.query.filter_by(activity_id=activity_id, status=EXPENSE_DRAFT).all()
        for exp in drafts:
            exp.status = EXPENSE_SUBMITTED
            exp.social_post_url = social_post_url

    db.session.commit()
    all_expenses = Expense.query.filter_by(activity_id=activity_id).all()
    return {"expenses": [e.to_dict() for e in all_expenses], "submitted": submit}
```

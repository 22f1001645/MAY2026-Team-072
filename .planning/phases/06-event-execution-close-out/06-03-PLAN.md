```python
def patch_report(activity_id, data, user):
    """FEEDBACK-01: Upsert draft report; Meetup Head submit."""
    activity = ActivityRepository.by_id(activity_id)
    if not activity:
        raise ApiError("Event not found", "not_found", 404)

    report = Report.query.filter_by(activity_id=activity_id).first()
    if not report:
        report = Report(activity_id=activity_id, summary="", status=REPORT_DRAFT)
        db.session.add(report)

    if "challenges" in data and data["challenges"] is not None:
        report.challenges = data["challenges"]
    if "learnings" in data and data["learnings"] is not None:
        report.learnings = data["learnings"]

    if data.get("status") == REPORT_SUBMITTED:
        is_head = ActivityTeamMember.query.filter_by(
            activity_id=activity_id, user_id=user.id, role=TEAM_ROLE_MEETUP_HEAD
        ).first()
        if not is_head and user.role != ROLE_ADMIN:
            raise ApiError("Only the Meetup Head can submit the reflection", "forbidden", 403)
        report.status = REPORT_SUBMITTED
        report.submitted_by = user.id
        report.submitted_at = datetime.utcnow()

    db.session.commit()
    return report
```
